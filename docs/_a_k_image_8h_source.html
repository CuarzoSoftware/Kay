66 bool setImage(std::shared_ptr<RImage> image) noexcept
67 {
68 if (image.get() == m_image.get())
69 return false;
70
71 /* Swapping between images is covered by the fine-grained damage the owner adds (e.g. a
72 * client's buffer damage), but gaining or losing the image changes whether the node draws
73 * AT ALL. That transition isn't covered by such damage: it may have been consumed by a
74 * frame where the node couldn't draw yet (null image), leaving the node permanently stale
75 * on the buffers composited before the image arrived. */
76 if ((m_image == nullptr) != (image == nullptr))
77 addDamage(AK_IRECT_INF);
78
79 addChange(CHImage);
80 m_image = image;
81 return true;
82 }
